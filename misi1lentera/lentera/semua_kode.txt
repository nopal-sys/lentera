"""

Endpoint:
    POST /api/misi1/soal        simpan hasil 1 soal   (dipanggil misi1.js saat soal dijawab benar)
                                body 'tipe': 'susun' (susun kalimat) atau 'baca' (baca kalimat, mikrofon)
    POST /api/misi1/level       simpan hasil 1 level  (dipanggil misi1.js saat popup hasil muncul)
    POST /api/pretest           simpan nilai pre-test anak (skala 0-100)

import sqlite3
from datetime import datetime, timedelta, timezone

from flask import Flask, g, jsonify, redirect, request, send_from_directory

BASE_DIR = os.path.dirname(os.path.abspath(__file__))
FRONTEND_DIR = os.path.join(BASE_DIR, 'misi1lentera')
DB_PATH = os.path.join(BASE_DIR, 'lentera.db')
WIB = timezone(timedelta(hours=7))  # tanggal laporan pakai waktu Indonesia Barat

# Frontend disajikan dari root: /level.html, /misi1.html, /gambar/xxx.jpg, dst
app = Flask(__name__, static_folder=FRONTEND_DIR, static_url_path='')


# =============================================================
#  ATURAN PENILAIAN (final)
#  Kalau diubah, ubah juga di misi1.js (hitungSkorSoal, skorSoal & tentukanPredikat)
#  dan baca-logika.js (BACA_POIN_*).
#
#  Satu level memuat dua jenis soal yang satu kesatuan:
#    susun : 100 / 75 / 50 / 25 menurut jumlah percobaan salah
#    baca  : 100 dikurangi 5 poin tiap bacaan salah (mikrofon bisa keliru mendengar)
# =============================================================
SKOR_PER_PERCOBAAN = [100, 75, 50]  # susun: benar di percobaan ke-1, ke-2, ke-3
SKOR_MINIMUM = 25                   # susun: percobaan ke-4 atau lebih

BACA_POIN_AWAL = 100                # baca: poin awal tiap kalimat
BACA_POIN_PER_SALAH = 5             # baca: pengurang tiap bacaan salah
BACA_POIN_MINIMUM = 0


def hitung_skor_soal(jumlah_salah, tipe='susun'):
    """susun: jumlah_salah 0 -> 100, 1 -> 75, 2 -> 50, >=3 -> 25.  baca: 100 - 5 x jumlah_salah."""
    if tipe == 'baca':
        return max(BACA_POIN_MINIMUM, BACA_POIN_AWAL - BACA_POIN_PER_SALAH * jumlah_salah)
    if jumlah_salah < len(SKOR_PER_PERCOBAAN):
        return SKOR_PER_PERCOBAAN[jumlah_salah]
    return SKOR_MINIMUM


def ambil_tipe(d):
    """Jenis soal dari browser: hanya 'baca' atau 'susun' (nilai lain dianggap susun)."""
    return 'baca' if str(d.get('tipe', '')).strip() == 'baca' else 'susun'


def tentukan_predikat(nilai):
    """Nilai 100 hanya mungkin kalau semua soal benar di percobaan 1 -> 'Sempurna'."""
    if nilai >= 100:
        return 'Sempurna'
    if nilai >= 90:
        return 'Sangat Baik'
    if nilai >= 75:
        return 'Baik'
    if nilai >= 60:
        return 'Cukup'
    return 'Perlu Pendampingan'


def rata_rata(daftar):
    return round(sum(daftar) / len(daftar), 1) if daftar else None


# =============================================================
#  IDENTITAS ANAK
#  Sekarang anak_id = ssoToken apa adanya. Kalau token berubah tiap login
#  atau punya mapping ke ID anak/nama, ubah di fungsi ini saja.
# =============================================================
def resolve_anak(sso_token):
    token = str(sso_token or '').strip()
    return token or None


# =============================================================
#  DATABASE (SQLite)
# =============================================================
SCHEMA = """
CREATE TABLE IF NOT EXISTS soal_hasil (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    anak_id      TEXT    NOT NULL,
    sesi_id      TEXT    NOT NULL,   -- 1 kali main 1 level = 1 sesi
    level        INTEGER NOT NULL,
    soal_id      TEXT    NOT NULL,
    jumlah_salah INTEGER NOT NULL,
    skor         INTEGER NOT NULL,
    waktu_detik  INTEGER NOT NULL,   -- data pendukung, TIDAK ikut nilai
    tanggal      TEXT    NOT NULL,
    UNIQUE (anak_id, sesi_id, soal_id)
);
CREATE TABLE IF NOT EXISTS level_hasil (
    id             INTEGER PRIMARY KEY AUTOINCREMENT,
    anak_id        TEXT    NOT NULL,
    sesi_id        TEXT    NOT NULL,
    level          INTEGER NOT NULL,
    nilai          REAL    NOT NULL,
    predikat       TEXT    NOT NULL,
    langsung_benar INTEGER NOT NULL,
    total_soal     INTEGER NOT NULL,
    main_ke        INTEGER NOT NULL, -- 1 = pertama kali level ini diselesaikan
    tanggal        TEXT    NOT NULL,
    UNIQUE (anak_id, sesi_id)
);
CREATE TABLE IF NOT EXISTS pretest (
    anak_id TEXT PRIMARY KEY,
    nilai   REAL NOT NULL,           -- skala 0-100
    tanggal TEXT NOT NULL
);
"""


def get_db():
    if 'db' not in g:
        g.db = sqlite3.connect(DB_PATH, timeout=10)
        g.db.row_factory = sqlite3.Row
    return g.db


@app.teardown_appcontext
def close_db(_exc):
    db = g.pop('db', None)
    if db is not None:
        db.close()


def init_db():
    with sqlite3.connect(DB_PATH) as db:
        db.executescript(SCHEMA)


def sekarang():
    return datetime.now(WIB).isoformat(timespec='seconds')


# =============================================================
#  HELPER VALIDASI
# =============================================================
class BadRequest(Exception):
    pass


@app.errorhandler(BadRequest)
def handle_bad_request(e):
    return jsonify(ok=False, error=str(e)), 400


def ambil_json():
    data = request.get_json(silent=True)
    if not isinstance(data, dict):
        raise BadRequest('Body harus JSON object.')
    return data


def ambil_int(data, kunci, minimum=0):
    try:
        nilai = int(data.get(kunci))
    except (TypeError, ValueError):
        raise BadRequest(f'"{kunci}" harus angka bulat.')
    if nilai < minimum:
        raise BadRequest(f'"{kunci}" minimal {minimum}.')
    return nilai


def ambil_teks(data, kunci):
    nilai = str(data.get(kunci) or '').strip()
    if not nilai or len(nilai) > 200:
        raise BadRequest(f'"{kunci}" wajib diisi (maks 200 karakter).')
    return nilai


def ambil_anak(data):
    anak_id = resolve_anak(data.get('ssoToken'))
    if not anak_id:
        raise BadRequest('"ssoToken" wajib diisi.')
    return anak_id


def simpan_soal(db, anak_id, sesi_id, level, soal_id, jumlah_salah, waktu_detik, tipe='susun'):
    """Upsert: kirim ulang soal yang sama di sesi yang sama tidak bikin data dobel."""
    skor = hitung_skor_soal(jumlah_salah, tipe)
    db.execute(
        """
        INSERT INTO soal_hasil (anak_id, sesi_id, level, soal_id, jumlah_salah, skor, waktu_detik, tanggal)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        ON CONFLICT (anak_id, sesi_id, soal_id) DO UPDATE SET
            jumlah_salah = excluded.jumlah_salah,
            skor         = excluded.skor,
            waktu_detik  = excluded.waktu_detik
        """,
        (anak_id, sesi_id, level, soal_id, jumlah_salah, skor, waktu_detik, sekarang()),
    )
    return skor

#  ENDPOINT: SIMPAN DATA

@app.post('/api/misi1/soal')
def api_simpan_soal():
    """Body: {ssoToken, sesi_id, level, soal_id, tipe ('susun'|'baca'), jumlah_salah, waktu_detik}"""
    d = ambil_json()
    anak_id = ambil_anak(d)
    db = get_db()
    skor = simpan_soal(
        db, anak_id, ambil_teks(d, 'sesi_id'), ambil_int(d, 'level', 1),
        ambil_teks(d, 'soal_id'), ambil_int(d, 'jumlah_salah'), ambil_int(d, 'waktu_detik'), ambil_tipe(d),
    )
    db.commit()
    return jsonify(ok=True, skor=skor)


@app.post('/api/misi1/level')
def api_simpan_level():
    """
    Body: {ssoToken, sesi_id, level, total_soal,
           soal: [{soal_id, tipe ('susun'|'baca'), jumlah_salah, waktu_detik}, ...]}
    Nilai dihitung ulang di server dari jumlah_salah (angka dari browser tidak dipercaya).
    Aman dikirim ulang (idempotent per sesi_id).
    """
    d = ambil_json()
    anak_id = ambil_anak(d)
    sesi_id = ambil_teks(d, 'sesi_id')
    level = ambil_int(d, 'level', 1)
    total_soal = ambil_int(d, 'total_soal', 1)
    soal = d.get('soal')
    if not isinstance(soal, list) or len({str(s.get('soal_id')) for s in soal if isinstance(s, dict)}) != total_soal:
        raise BadRequest('Jumlah data "soal" harus sama dengan total_soal (dan soal_id tidak boleh dobel).')

    db = get_db()
    skor_list, langsung_benar = [], 0
    for s in soal:
        salah = ambil_int(s, 'jumlah_salah')
        skor_list.append(simpan_soal(db, anak_id, sesi_id, level, ambil_teks(s, 'soal_id'), salah, ambil_int(s, 'waktu_detik'), ambil_tipe(s)))
        if salah == 0:
            langsung_benar += 1

    nilai = rata_rata(skor_list)
    predikat = tentukan_predikat(nilai)

    sudah = db.execute('SELECT main_ke FROM level_hasil WHERE anak_id = ? AND sesi_id = ?', (anak_id, sesi_id)).fetchone()
    if sudah:
        main_ke = sudah['main_ke']
        db.execute(
            'UPDATE level_hasil SET nilai=?, predikat=?, langsung_benar=?, total_soal=? WHERE anak_id=? AND sesi_id=?',
            (nilai, predikat, langsung_benar, total_soal, anak_id, sesi_id),
        )
    else:
        n = db.execute('SELECT COUNT(*) AS n FROM level_hasil WHERE anak_id = ? AND level = ?', (anak_id, level)).fetchone()['n']
        main_ke = n + 1
        db.execute(
            """INSERT INTO level_hasil (anak_id, sesi_id, level, nilai, predikat, langsung_benar, total_soal, main_ke, tanggal)
               VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
            (anak_id, sesi_id, level, nilai, predikat, langsung_benar, total_soal, main_ke, sekarang()),
        )
    db.commit()
    return jsonify(ok=True, nilai=nilai, predikat=predikat, langsung_benar=langsung_benar,
                   total_soal=total_soal, main_ke=main_ke, dihitung_di_laporan=(main_ke == 1))


@app.post('/api/pretest')
def api_simpan_pretest():
    """
    Body: {ssoToken, nilai}            nilai sudah skala 0-100
      atau {ssoToken, benar, total}    otomatis diubah ke skala 0-100
    """
    d = ambil_json()
    anak_id = ambil_anak(d)
    if 'benar' in d and 'total' in d:
        total = ambil_int(d, 'total', 1)
        benar = ambil_int(d, 'benar')
        if benar > total:
            raise BadRequest('"benar" tidak boleh lebih besar dari "total".')
        nilai = round(benar / total * 100, 1)
    else:
        try:
            nilai = round(float(d.get('nilai')), 1)
        except (TypeError, ValueError):
            raise BadRequest('Isi "nilai" (0-100) atau "benar" + "total".')
        if not 0 <= nilai <= 100:
            raise BadRequest('"nilai" harus di antara 0 dan 100.')
    db = get_db()
    db.execute(
        'INSERT INTO pretest (anak_id, nilai, tanggal) VALUES (?, ?, ?) '
        'ON CONFLICT (anak_id) DO UPDATE SET nilai = excluded.nilai, tanggal = excluded.tanggal',
        (anak_id, nilai, sekarang()),
    )
    db.commit()
    return jsonify(ok=True, anak_id=anak_id, nilai=nilai)


# =============================================================
#  LAPORAN
#  Yang dihitung ke nilai = PENYELESAIAN PERTAMA tiap level (main_ke = 1).
#  "Main Lagi" tetap tersimpan, tapi tidak mengubah nilai laporan,
#  supaya anak tidak bisa mengulang sampai 100.
# =============================================================
def susun_laporan(db, anak_id):
    pre = db.execute('SELECT nilai FROM pretest WHERE anak_id = ?', (anak_id,)).fetchone()
    semua = db.execute('SELECT * FROM level_hasil WHERE anak_id = ? ORDER BY level, main_ke', (anak_id,)).fetchall()
    ada_soal = db.execute('SELECT 1 FROM soal_hasil WHERE anak_id = ? LIMIT 1', (anak_id,)).fetchone()
    if not pre and not semua and not ada_soal:
        return None

    pertama = [r for r in semua if r['main_ke'] == 1]
    ulang = {}
    for r in semua:
        if r['main_ke'] > 1:
            ulang[r['level']] = ulang.get(r['level'], 0) + 1

    nilai_literasi = rata_rata([r['nilai'] for r in pertama])
    pretest = pre['nilai'] if pre else None
    perkembangan = round(nilai_literasi - pretest, 1) if (nilai_literasi is not None and pretest is not None) else None

    total_soal = sum(r['total_soal'] for r in pertama)
    soal_langsung = sum(r['langsung_benar'] for r in pertama)

    per_tanggal = {}
    for r in sorted(pertama, key=lambda r: r['tanggal']):
        per_tanggal.setdefault(r['tanggal'][:10], []).append(r['nilai'])

    return {
        'anak_id': anak_id,
        'nilai_literasi': nilai_literasi,
        'predikat': tentukan_predikat(nilai_literasi) if nilai_literasi is not None else None,
        'pretest': pretest,
        'perkembangan': perkembangan,
        'soal_langsung_benar': soal_langsung,
        'total_soal': total_soal,
        'persen_langsung_benar': round(soal_langsung / total_soal * 100, 1) if total_soal else None,
        'level_selesai': [
            {
                'level': r['level'], 'nilai': r['nilai'], 'predikat': r['predikat'],
                'langsung_benar': r['langsung_benar'], 'total_soal': r['total_soal'],
                'tanggal': r['tanggal'][:10], 'main_ulang': ulang.get(r['level'], 0),
            }
            for r in pertama
        ],
        'tren': [{'tanggal': t, 'nilai': rata_rata(v)} for t, v in per_tanggal.items()],
    }


@app.get('/api/laporan/<path:anak_id>')
def api_laporan_anak(anak_id):
    hasil = susun_laporan(get_db(), anak_id)
    if hasil is None:
        return jsonify(ok=False, error='Data anak tidak ditemukan.'), 404
    return jsonify(ok=True, **hasil)


@app.get('/api/laporan')
def api_laporan_semua():
    db = get_db()
    ids = [r['anak_id'] for r in db.execute(
        'SELECT anak_id FROM level_hasil UNION SELECT anak_id FROM pretest UNION SELECT anak_id FROM soal_hasil ORDER BY anak_id'
    ).fetchall()]
    ringkas = []
    for anak_id in ids:
        l = susun_laporan(db, anak_id)
        ringkas.append({k: l[k] for k in ('anak_id', 'nilai_literasi', 'predikat', 'pretest', 'perkembangan')}
                       | {'jumlah_level': len(l['level_selesai'])})
    return jsonify(ok=True, anak=ringkas)

'use strict';

// =============================================================
// LENTERA — GAME BACA KALIMAT : LOGIKA MURNI (tanpa DOM)
// Dipakai baca.js di browser, dan bisa diuji di Node:
//     node -e "console.log(require('./baca-logika.js'))"
// =============================================================

(function (root, factory) {
  if (typeof module === 'object' && module.exports) {
    module.exports = factory();
  } else {
    root.BacaLogika = factory();
  }
})(typeof self !== 'undefined' ? self : this, function () {

  // ===========================================================
  // ATURAN PENILAIAN (final)
  // Tiap kalimat bernilai 100. Setiap bacaan yang salah
  // mengurangi 5 poin. Poin tidak pernah di bawah 0.
  // Kalau diubah, ubah juga di app.py (BACA_POIN_*).
  // ===========================================================
  const POIN_AWAL = 100;
  const POIN_PER_SALAH = 5;
  const POIN_MINIMUM = 0;

  function hitungPoin(jumlahSalah) {
    const salah = Math.max(0, Math.floor(Number(jumlahSalah) || 0));
    return Math.max(POIN_MINIMUM, POIN_AWAL - POIN_PER_SALAH * salah);
  }

  // Sama dengan Game 1 (misi1.js) supaya predikat & bintang seragam.
  function tentukanPredikat(nilai) {
    if (nilai >= 100) return 'Sempurna';
    if (nilai >= 90) return 'Sangat Baik';
    if (nilai >= 75) return 'Baik';
    if (nilai >= 60) return 'Cukup';
    return 'Perlu Pendampingan';
  }

  function hitungBintang(nilai) {
    if (nilai < 50) return 1;
    if (nilai < 80) return 2;
    return 3;
  }

  function rataRata(daftar) {
    if (!daftar.length) return 0;
    const jumlah = daftar.reduce((total, n) => total + n, 0);
    return Math.round((jumlah / daftar.length) * 10) / 10;
  }

  // ===========================================================
  // ANGKA -> KATA  (mic sering menulis "7" untuk "tujuh")
  // ===========================================================
  const SATUAN = [
    'nol', 'satu', 'dua', 'tiga', 'empat',
    'lima', 'enam', 'tujuh', 'delapan', 'sembilan'
  ];

  function angkaKeKata(n) {
    n = Math.floor(n);

    if (n < 10) return SATUAN[n];
    if (n === 10) return 'sepuluh';
    if (n === 11) return 'sebelas';

    if (n < 20) return `${SATUAN[n - 10]} belas`;

    if (n < 100) {
      const sisa = n % 10;
      return `${SATUAN[Math.floor(n / 10)]} puluh` +
        (sisa ? ` ${SATUAN[sisa]}` : '');
    }

    if (n < 1000) {
      const ratusan = Math.floor(n / 100);
      const sisa = n % 100;
      const awal = ratusan === 1 ? 'seratus' : `${SATUAN[ratusan]} ratus`;
      return awal + (sisa ? ` ${angkaKeKata(sisa)}` : '');
    }

    if (n < 1000000) {
      const ribuan = Math.floor(n / 1000);
      const sisa = n % 1000;
      const awal = ribuan === 1 ? 'seribu' : `${angkaKeKata(ribuan)} ribu`;
      return awal + (sisa ? ` ${angkaKeKata(sisa)}` : '');
    }

    return String(n);
  }

  // ===========================================================
  // NORMALISASI
  // huruf kecil, tanpa aksen, angka jadi kata, tanda baca jadi spasi
  // ===========================================================
  function normalisasi(teks) {
    let s = String(teks === null || teks === undefined ? '' : teks)
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '');

    s = s.replace(/\d+/g, angka => ` ${angkaKeKata(parseInt(angka, 10))} `);
    s = s.replace(/[^a-z\s]/g, ' ');

    return s.replace(/\s+/g, ' ').trim();
  }

  function pecahKata(teks) {
    const bersih = normalisasi(teks);
    return bersih ? bersih.split(' ') : [];
  }

  // ===========================================================
  // KUNCI FONETIK
  // Ejaan Indonesia hampir sama dengan bunyinya, tetapi mic
  // sering memilih ejaan lain untuk nama: Bela/Bella, Andi/Andy,
  // Aziz/Azis, Raka/Rakha, Refan/Revan. Kata yang bunyinya sama
  // diberi kunci yang sama. Kata yang benar-benar beda
  // (membeli/memberi, sayur/sayuran) tetap dianggap beda.
  // ===========================================================
  function kunciFonetik(kata) {
    return String(kata)
      .replace(/kh/g, 'k')
      .replace(/ph/g, 'f')
      .replace(/th/g, 't')
      .replace(/sh/g, 's')
      .replace(/sy/g, 's')
      .replace(/dj/g, 'j')
      .replace(/tj/g, 'c')
      .replace(/oe/g, 'u')
      .replace(/x/g, 'ks')
      .replace(/q/g, 'k')
      .replace(/v/g, 'f')
      .replace(/z/g, 's')
      .replace(/([a-z])\1+/g, '$1')
      .replace(/y$/, 'i');
  }

  // ===========================================================
  // PERIKSA BACAAN
  //
  // Bacaan dianggap SAMA bila ada satu rentang kata berurutan
  // pada hasil mic yang hurufnya persis sama dengan kalimat
  // target (spasi tidak dihitung, jadi "perpus takaan" tetap
  // cocok dengan "perpustakaan").
  //
  // Karena yang dicari adalah rentang, kata tambahan di awal
  // atau di akhir ("eh", "ya") dan bacaan yang diulang setelah
  // salah ucap ("ibu masak ibu memasak nasi") tetap benar.
  // Kata yang kurang, tertukar, atau diganti kata lain: salah.
  //
  // Hasil:
  //   benar        true bila kalimat sama
  //   cocok[i]     kata target ke-i sudah terdengar (berurutan)
  //   jumlahCocok  banyaknya kata target yang terdengar
  //   jumlahKata   banyaknya kata target
  //   kataTerdengar banyaknya kata yang ditangkap mic
  // ===========================================================
  function periksaBacaan(target, hasil) {
    const kataTarget = pecahKata(target).map(kunciFonetik);
    const kataHasil = pecahKata(hasil).map(kunciFonetik);

    const cocok = new Array(kataTarget.length).fill(false);

    if (kataTarget.length === 0) {
      return {
        benar: false,
        cocok,
        jumlahCocok: 0,
        jumlahKata: 0,
        kataTerdengar: kataHasil.length
      };
    }

    const targetRapat = kataTarget.join('');
    let benar = false;

    for (let awal = 0; awal < kataHasil.length && !benar; awal++) {
      let gabungan = '';

      for (let akhir = awal; akhir < kataHasil.length; akhir++) {
        gabungan += kataHasil[akhir];

        if (gabungan.length > targetRapat.length) break;

        if (gabungan === targetRapat) {
          benar = true;
          break;
        }
      }
    }

    // Status per kata untuk sorotan: pencocokan berurutan.
    let posisi = 0;

    for (let i = 0; i < kataTarget.length; i++) {
      for (let j = posisi; j < kataHasil.length; j++) {
        if (kataHasil[j] === kataTarget[i]) {
          cocok[i] = true;
          posisi = j + 1;
          break;
        }
      }
    }

    if (benar) cocok.fill(true);

    return {
      benar,
      cocok,
      jumlahCocok: cocok.filter(Boolean).length,
      jumlahKata: kataTarget.length,
      kataTerdengar: kataHasil.length
    };
  }

  // Mic memberi beberapa kemungkinan kalimat. Benar bila salah
  // satunya cocok; kalau tidak, pakai yang paling banyak cocok.
  function periksaAlternatif(target, daftarTeks) {
    let terbaik = null;
    let teksTerbaik = '';

    daftarTeks.forEach(teks => {
      const hasil = periksaBacaan(target, teks);

      if (
        !terbaik ||
        (hasil.benar && !terbaik.benar) ||
        (
          hasil.benar === terbaik.benar &&
          hasil.jumlahCocok > terbaik.jumlahCocok
        )
      ) {
        terbaik = hasil;
        teksTerbaik = teks;
      }
    });

    return {
      ...(terbaik || periksaBacaan(target, '')),
      teks: teksTerbaik
    };
  }

  // ===========================================================
  // KATA UNTUK TAMPILAN
  // Kalimat ditampilkan per kata (dengan tanda bacanya). Satu
  // kata tampilan bisa berisi 0 sampai beberapa kata ternormalisasi
  // ("murid-murid" = 2, "7" = "tujuh", "," = 0), jadi dibuat peta
  // dari kata tampilan ke indeks pada hasil periksaBacaan().
  // ===========================================================
  function pecahTampilan(kalimat) {
    const kata = String(kalimat || '').trim().split(/\s+/).filter(Boolean);
    let nomor = 0;

    const peta = kata.map(item => {
      const jumlah = pecahKata(item).length;
      const indeks = [];

      for (let i = 0; i < jumlah; i++) indeks.push(nomor++);

      return indeks;
    });

    return { kata, peta };
  }

  // 'benar' | 'salah' untuk tiap kata tampilan.
  // Kata tanpa huruf (mis. tanda "-") dianggap benar.
  function statusTampilan(peta, cocok) {
    return peta.map(indeks => (
      indeks.every(i => cocok[i]) ? 'benar' : 'salah'
    ));
  }

  return {
    POIN_AWAL,
    POIN_PER_SALAH,
    POIN_MINIMUM,
    hitungPoin,
    tentukanPredikat,
    hitungBintang,
    rataRata,
    angkaKeKata,
    normalisasi,
    pecahKata,
    kunciFonetik,
    periksaBacaan,
    periksaAlternatif,
    pecahTampilan,
    statusTampilan
  };
});

  // Gabungkan semua hasil sesi ini menjadi kemungkinan kalimat.
  function prosesHasil(sesi, event) {
    if (sesi.selesai || sesi !== sesiAktif) return;

    const hasil = event.results;

    if (!hasil || hasil.length === 0) return;

    let awal = '';

    for (let i = 0; i < hasil.length - 1; i++) {
      awal += ` ${hasil[i][0].transcript}`;
    }

    const terakhir = hasil[hasil.length - 1];
    const kandidat = [];

    for (let k = 0; k < terakhir.length; k++) {
      kandidat.push(`${awal} ${terakhir[k].transcript}`.trim());
    }

    const utama = kandidat[0] || '';

    sesi.teks = utama;
    sesi.kandidat = kandidat;

    if (utama) sesi.adaSuara = true;

    tampilkanTerdengar(utama);

    const periksa = L.periksaAlternatif(soal.kalimat, kandidat);

    // Sorot kata yang sudah terdengar sambil anak membaca.
    terapkanSorotan(periksa.cocok, { berikut: true });

    // Kalimat sudah sama: langsung selesai, tidak perlu menunggu diam.
    if (periksa.benar) {
      sesi.selesai = true;
      bersihkanPewaktu(sesi);
      sesiAktif = null;

      try {
        sesi.rec.abort();
      } catch (_) {
        // sudah berhenti
      }

      nilaiBacaan(periksa, periksa.teks);
      return;
    }

    if (utama) jadwalkanSenyap(sesi);
  }

  // Dipanggil saat mikrofon berhenti (diam, diketuk, atau error).
  function akhiriSesi(sesi) {

    const teks = sesi.teks.trim();

    // Masalah mikrofon / jaringan bukan kesalahan anak: poin aman.
    if (GANGGUAN.includes(sesi.error)) {
      terapkanSorotan(null);
      setMode('siap');
      statusLine('');

      terapkanSorotan(null);
      setMode('siap');

        'Belum ada suara yang terdengar. Dekatkan mikrofon, lalu tekan ' +
        'mikrofon dan baca lagi. Poinmu tidak berkurang.'
      );

      return;

    const periksa = L.periksaAlternatif(soal.kalimat, sesi.kandidat.length ? sesi.kandidat : [teks]);

    nilaiBacaan(periksa, periksa.teks || teks);

  function polaBacaan(periksa) {
    if (periksa.benar) return '';

    return periksa.kataTerdengar < periksa.jumlahKata
      ? 'baca_kata_terlewat'

  function nilaiBacaan(periksa, teksTerdengar, opsi = {}) {
    const waktuMs = Math.max(0, Date.now() - mulaiPercobaan);

    setMode('dialog');

    } else {
      terapkanSorotan(periksa.cocok, { tandaiSalah: true });

    onHasil({
      benar: Boolean(periksa.benar),
      teks: teksTerdengar,
      waktuMs,
      pola: polaBacaan(periksa),
      pakaiContoh: Boolean(soal && pakaiContoh.has(soal.id)),
      bantuanGuru: Boolean(opsi.bantuanGuru),
      jumlahKata: periksa.jumlahKata,
      jumlahCocok: periksa.jumlahCocok,

    ucapan.lang = CFG.bahasa;
    ucapan.rate = CFG.lajuContoh;

      clearTimeout(pengaman);

      sedangBicara = false;
      statusLine('Sekarang giliranmu. Tekan mikrofon, lalu baca kalimatnya.');
      perbaruiTombol();

    const pengaman = setTimeout(selesaiBicara, 20000);

    ucapan.onend = selesaiBicara;
    ucapan.onerror = selesaiBicara;

    // Mendengar contoh = memakai petunjuk. Dicatat ke Dashboard.
    pakaiContoh.add(soal.id);

  // Perangkat tanpa mikrofon: guru menandai bacaan benar supaya level tidak macet.
  function bantuanGuru() {

    batalkan();

    const periksa = {

      jumlahCocok: tampilan.kata.length,

function grafikTren(tren) {
  if (!tren.length) return '<p class="text-sm text-slate-400">Belum ada data.</p>';

  const W = 640, H = 260, L = 44, R = 24, T = 20, B = 46;
  const innerW = W - L - R, innerH = H - T - B;
  const x = (i) => (tren.length === 1 ? L + innerW / 2 : L + (innerW * i) / (tren.length - 1));
  const y = (v) => T + innerH - (innerH * v) / 100;

  const grid = [0, 25, 50, 75, 100].map((v) => `
    <line x1="${L}" y1="${y(v)}" x2="${W - R}" y2="${y(v)}" stroke="#e2e8f0" stroke-width="1" />
    <text x="${L - 8}" y="${y(v) + 4}" text-anchor="end" font-size="11" fill="#94a3b8">${v}</text>`).join('');

  const garis = tren.length > 1
    ? `<polyline fill="none" stroke="#f97316" stroke-width="3" stroke-linejoin="round" stroke-linecap="round"
         points="${tren.map((t, i) => `${x(i)},${y(t.nilai)}`).join(' ')}" />`
    : '';

  const titik = tren.map((t, i) => {
    const [, m, d] = t.tanggal.split('-');
    return `
      <circle cx="${x(i)}" cy="${y(t.nilai)}" r="6" fill="#ffffff" stroke="#f97316" stroke-width="3" />
      <text x="${x(i)}" y="${y(t.nilai) - 12}" text-anchor="middle" font-size="12" font-weight="700" fill="#c2410c">${angka(t.nilai)}</text>
      <text x="${x(i)}" y="${H - B + 20}" text-anchor="middle" font-size="11" fill="#64748b">${parseInt(d, 10)} ${NAMA_BULAN[parseInt(m, 10) - 1]}</text>`;
  }).join('');

  return `
    <svg viewBox="0 0 ${W} ${H}" class="w-full h-auto" role="img" aria-label="Grafik tren nilai per tanggal">
      ${grid}${garis}${titik}
    </svg>
    ${tren.length === 1 ? '<p class="text-xs text-slate-400 mt-1">Baru ada 1 tanggal. Garis muncul setelah ada nilai di tanggal lain.</p>' : ''}`;
}

function decodeJwtPayload(token) {
    try {
        const parts = token.split('.');
        if (parts.length !== 3) return null;

        let payload = parts[1]
            .replace(/-/g, '+')
            .replace(/_/g, '/');

        payload += '='.repeat((4 - payload.length % 4) % 4);

        const binary = atob(payload);
        const bytes = Uint8Array.from(
            binary,
            character => character.charCodeAt(0)
        );

        return JSON.parse(new TextDecoder().decode(bytes));
    } catch (error) {
        console.error('Token Dashboard tidak bisa dibaca:', error);
        return null;
    }
}

const videoElement = document.getElementById('webcam');
const gestureStatusEl = document.getElementById('gesture-status');
const cameraDotEl = document.getElementById('camera-dot');
const virtualCursorEl = document.getElementById('virtual-cursor');

const CFG = {
    detectConf: 0.6,
    trackConf: 0.5,
    smooth: 0.5,
    edgeMargin: 0.08,
    lostGraceFrames: 20,
    hitRadius: 20,
    dwellMs: 1200,
    lockDistance: 60
};

let smoothX = null;
let smoothY = null;
let lostFrames = 0;

let hoverEl = null;
let dwellStart = 0;
let dwellLock = null;

class OneEuro {
    constructor() {
        this.reset();
    }

    reset() {
        this.xPrev = null;
        this.dxPrev = 0;
        this.tPrev = 0;
    }

    static alpha(cutoff, dt) {
        const tau = 1 / (2 * Math.PI * cutoff);
        return 1 / (1 + tau / dt);
    }

    filter(x, t) {
        if (this.xPrev === null) {
            this.xPrev = x;
            this.dxPrev = 0;
            this.tPrev = t;

            return x;
        }

        const dt = Math.max(
            (t - this.tPrev) / 1000,
            0.001
        );

        this.tPrev = t;

        const dx = (x - this.xPrev) / dt;

        this.dxPrev +=
            OneEuro.alpha(1, dt) *
            (dx - this.dxPrev);

        const cutoff = 0.8 + 0.05 * Math.abs(this.dxPrev);

        this.xPrev +=
            OneEuro.alpha(cutoff, dt) *
            (x - this.xPrev);

        return this.xPrev;
    }
}

const cursorXFilter = new OneEuro();
const cursorYFilter = new OneEuro();

const handCanvas = document.createElement('canvas');

handCanvas.style.cssText =
    'position:fixed;left:0;top:0;pointer-events:none;' +
    'z-index:2147483000;display:none;';

document.body.appendChild(handCanvas);

const handCtx = handCanvas.getContext('2d');

const HAND_CONNECTIONS = [
    [0, 1], [1, 2], [2, 3], [3, 4],
    [0, 5], [5, 6], [6, 7], [7, 8],
    [5, 9], [9, 10], [10, 11], [11, 12],
    [9, 13], [13, 14], [14, 15], [15, 16],
    [13, 17], [17, 18], [18, 19], [19, 20],
    [0, 17]
];

const FINGER_TIPS = [4, 8, 12, 16, 20];

function drawHand(landmarks, toScreen) {
    const dpr = window.devicePixelRatio || 1;
    const width = window.innerWidth;
    const height = window.innerHeight;

    if (
        handCanvas.width !== Math.round(width * dpr) ||
        handCanvas.height !== Math.round(height * dpr)
    ) {
        handCanvas.width = Math.round(width * dpr);
        handCanvas.height = Math.round(height * dpr);

        handCanvas.style.width = `${width}px`;
        handCanvas.style.height = `${height}px`;
    }

    handCanvas.style.display = 'block';

    handCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
    handCtx.clearRect(0, 0, width, height);

    handCtx.lineCap = 'round';
    handCtx.lineJoin = 'round';
    handCtx.lineWidth = 4;
    handCtx.strokeStyle = '#60a5fa';

    handCtx.beginPath();

    HAND_CONNECTIONS.forEach(([a, b]) => {
        const p = toScreen(landmarks[a]);
        const q = toScreen(landmarks[b]);

        handCtx.moveTo(p.x, p.y);
        handCtx.lineTo(q.x, q.y);
    });

    handCtx.stroke();

    for (let i = 0; i < landmarks.length; i++) {
        const p = toScreen(landmarks[i]);
        const isTip = FINGER_TIPS.includes(i);

        handCtx.beginPath();

        handCtx.arc(
            p.x,
            p.y,
            isTip ? 11 : 6,
            0,
            Math.PI * 2
        );

        handCtx.fillStyle =
            i === 8 ? '#f97316' : '#ffffff';

        handCtx.fill();

        handCtx.lineWidth = 2;
        handCtx.strokeStyle = '#3b82f6';

        handCtx.stroke();
    }
}

function setCursorProgress(progress) {
    if (!virtualCursorEl) return;

    const px = Math.max(0, Math.min(1, progress)) * 10;

    virtualCursorEl.style.boxShadow =
        px > 0
            ? `0 0 0 ${px}px rgba(249,115,22,0.35)`
            : '';
}

function resetDwell() {
    if (hoverEl) {
        hoverEl.classList.remove('cursor-hover');
    }

    hoverEl = null;
    dwellStart = 0;

    setCursorProgress(0);
}

function cariTombolDekat(x, y) {
    const radius = CFG.hitRadius;

    const offsets = [
        [0, 0],
        [radius, 0],
        [-radius, 0],
        [0, radius],
        [0, -radius]
    ];

    for (const [dx, dy] of offsets) {
        const element = document.elementFromPoint(
            x + dx,
            y + dy
        );

        const target = element
            ? element.closest('.interactive-elem')
            : null;

        if (target) return target;
    }

    return null;
}

function updateDwell(x, y, now) {
    if (dwellLock) {
        const distance = Math.hypot(
            x - dwellLock.x,
            y - dwellLock.y
        );

        if (distance < CFG.lockDistance) {
            resetDwell();
            return;
        }

        dwellLock = null;
    }

    const target = cariTombolDekat(x, y);

    if (!target) {
        resetDwell();
        return;
    }

    if (target !== hoverEl) {
        resetDwell();

        hoverEl = target;
        hoverEl.classList.add('cursor-hover');

        dwellStart = now;
        return;
    }

    const progress = (now - dwellStart) / CFG.dwellMs;

    setCursorProgress(progress);

    if (progress >= 1) {
        const clickTarget = hoverEl;

        dwellLock = { x, y };

        resetDwell();

        clickTarget.click();
    }
}

function onResults(results) {
    const now = performance.now();

    const daftarTangan = results.multiHandLandmarks;

    if (!daftarTangan || daftarTangan.length === 0) {
        lostFrames++;

        if (lostFrames > CFG.lostGraceFrames) {
            if (gestureStatusEl) {
                gestureStatusEl.textContent = 'Mencari Tangan...';
            }

            if (cameraDotEl) {
                cameraDotEl.className =
                    'w-1.5 h-1.5 rounded-full ' +
                    'bg-amber-400 animate-ping';
            }

            if (virtualCursorEl) {
                virtualCursorEl.style.display = 'none';
            }

            handCanvas.style.display = 'none';

            resetDwell();

            smoothX = null;
            smoothY = null;

            cursorXFilter.reset();
            cursorYFilter.reset();
        }

        return;
    }

    lostFrames = 0;

    const raw = daftarTangan[0];
    const indexTip = raw[8];

    const margin = CFG.edgeMargin;

    const targetX =
        Math.min(
            1,
            Math.max(
                0,
                ((1 - indexTip.x) - margin) / (1 - 2 * margin)
            )
        ) * window.innerWidth;

    const targetY =
        Math.min(
            1,
            Math.max(
                0,
                (indexTip.y - margin) / (1 - 2 * margin)
            )
        ) * window.innerHeight;

    if (smoothX === null) {
        smoothX = targetX;
        smoothY = targetY;
    } else {
        smoothX += (targetX - smoothX) * CFG.smooth;
        smoothY += (targetY - smoothY) * CFG.smooth;
    }

    smoothX = cursorXFilter.filter(smoothX, now);
    smoothY = cursorYFilter.filter(smoothY, now);

    if (gestureStatusEl) {
        gestureStatusEl.textContent = 'Telunjuk Aktif 👆';
    }

    if (cameraDotEl) {
        cameraDotEl.className =
            'w-1.5 h-1.5 rounded-full ' +
            'bg-emerald-500 animate-pulse';
    }

    if (virtualCursorEl) {
        virtualCursorEl.style.display = 'flex';
        virtualCursorEl.style.left = `${smoothX}px`;
        virtualCursorEl.style.top = `${smoothY}px`;
        virtualCursorEl.textContent = '👆';
    }

    const toScreen = point => ({
        x:
            Math.min(
                1,
                Math.max(
                    0,
                    ((1 - point.x) - margin) / (1 - 2 * margin)
                )
            ) * window.innerWidth,

        y:
            Math.min(
                1,
                Math.max(
                    0,
                    (point.y - margin) / (1 - 2 * margin)
                )
            ) * window.innerHeight
    });

    drawHand(raw, toScreen);
    updateDwell(smoothX, smoothY, now);
}

function mulaiKamera() {
    if (
        typeof Hands === 'undefined' ||
        typeof Camera === 'undefined' ||
        !videoElement
    ) {
        if (gestureStatusEl) {
            gestureStatusEl.textContent = 'Mode Sentuh Aktif';
        }

        console.warn(
            'MediaPipe atau elemen webcam tidak tersedia. ' +
            'Kartu level tetap dapat dipilih dengan mouse.'
        );

        return;
    }

    try {
        const hands = new Hands({
            locateFile: file =>
                `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
        });

        hands.setOptions({
            maxNumHands: 1,
            modelComplexity: 1,
            minDetectionConfidence: CFG.detectConf,
            minTrackingConfidence: CFG.trackConf
        });

        hands.onResults(onResults);

        const camera = new Camera(videoElement, {
            onFrame: async () => {
                await hands.send({
                    image: videoElement
                });
            },
            width: 640,
            height: 480
        });

        camera.start().catch(error => {
            console.error('Kamera gagal dijalankan:', error);

            if (gestureStatusEl) {
                gestureStatusEl.textContent = 'Mode Sentuh Aktif';
            }

            if (cameraDotEl) {
                cameraDotEl.className =
                    'w-1.5 h-1.5 rounded-full bg-rose-500';
            }
        });

    } catch (error) {
        console.error('Gagal memulai MediaPipe:', error);

        if (gestureStatusEl) {
            gestureStatusEl.textContent = 'Mode Sentuh Aktif';
        }
    }
}

const urlParams = new URLSearchParams(window.location.search);
const ssoToken = urlParams.get('ssoToken') || '';
const legacySsoToken = ssoToken || 'DEMO_TOKEN_123';

function decodeJwtPayload(token) {
  try {
    const part = String(token || '').split('.')[1];
    if (!part) return null;

    const normalized = part.replace(/-/g, '+').replace(/_/g, '/');
    const padded = normalized + '='.repeat((4 - normalized.length % 4) % 4);
    const binary = atob(padded);
    const bytes = Uint8Array.from(binary, ch => ch.charCodeAt(0));

    return JSON.parse(new TextDecoder().decode(bytes));
  } catch (error) {
    console.warn('Token Dashboard tidak dapat dibaca:', error);
    return null;
  }
}

// ============================================================
// SUSUN KALIMAT + BACA KALIMAT = SATU KESATUAN
//
// Satu level Literasi memuat dua jenis soal dalam satu game:
//   tipe 'susun' : susun kata menjadi kalimat (kartu kata)
//   tipe 'baca'  : baca kalimat dengan suara (mikrofon), kompetensi LT-KETEPATAN
// Mode Dashboard: komposisi ditentukan server (3 susun, lalu 2 baca).
// Mode mandiri : komposisi dibuat di sini dari soal-misi1.js + soal-baca.js.
// ============================================================

const SOAL_PER_LEVEL = 5;
const SOAL_BACA_PER_LEVEL = 2;
const KOMPETENSI_BACA = 'LT-KETEPATAN';

function acakDaftar(daftar) {
  const salinan = [...daftar];

  for (let i = salinan.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));

    [salinan[i], salinan[j]] = [salinan[j], salinan[i]];
  }

  return salinan;
}

function soalLevelMandiri(level) {
  const susun = questions
    .filter(q => Number(q.level) === level)
    .map(q => ({ ...q, tipe: 'susun' }));

  const bankBaca = typeof soalBaca !== 'undefined' ? soalBaca : [];

  const baca = bankBaca
    .filter(b => Number(b.level) === level)
    .map(b => ({
      id: b.id,
      tipe: 'baca',
      level: b.level,
      gambar: b.gambar,
      kalimat: b.kalimat,
      kompetensiDashboard: KOMPETENSI_BACA
    }));

  if (baca.length === 0 || susun.length === 0) {
    return [...susun, ...baca].slice(0, SOAL_PER_LEVEL);
  }

  const jumlahBaca = Math.min(baca.length, SOAL_BACA_PER_LEVEL, SOAL_PER_LEVEL - 1);
  const jumlahSusun = Math.min(susun.length, SOAL_PER_LEVEL - jumlahBaca);

  // Soal susun lebih dulu (urutan aslinya), lalu soal baca.
  const terpilihSusun = acakDaftar(susun)
    .slice(0, jumlahSusun)
    .sort((a, b) => susun.indexOf(a) - susun.indexOf(b));

  return [
    ...terpilihSusun,
    ...acakDaftar(baca).slice(0, Math.min(baca.length, SOAL_PER_LEVEL - jumlahSusun))
  ];
}

// Kompetensi sebuah jawaban: bacaan mikrofon selalu LT-KETEPATAN; soal susun tidak
// pernah dihitung sebagai ketepatan membaca (server juga menjaga aturan ini).
function kompetensiEvent(q) {
  if (q.tipe === 'baca') return KOMPETENSI_BACA;

  const dasar = hasDashboardSession
    ? dashboardCompetency
    : normalizeCompetency(q.kompetensiDashboard || q.kompetensi || '');

  if (dasar === KOMPETENSI_BACA) {
    const sendiri = normalizeCompetency(q.kompetensiDashboard || '');

    return sendiri && sendiri !== KOMPETENSI_BACA ? sendiri : 'LT-KALIMAT';
  }

  return dasar;
}

function normalizeCompetency(code) {
  const value = String(code || '').trim();

  const aliases = {
    menyusun_kalimat: 'LT-KALIMAT',
    MENYUSUN_KALIMAT: 'LT-KALIMAT',
    kosakata: 'LT-KOSAKATA',
    KOSAKATA: 'LT-KOSAKATA',
    membaca_lancar: 'LT-LANCAR',
    MEMBACA_LANCAR: 'LT-LANCAR',
    ketepatan: 'LT-KETEPATAN',
    KETEPATAN: 'LT-KETEPATAN',
    baca_lantang: 'LT-KETEPATAN',
    BACA_LANTANG: 'LT-KETEPATAN',
    baca_kalimat: 'LT-KETEPATAN',
    BACA_KALIMAT: 'LT-KETEPATAN'
  };

  return aliases[value] || value;
}

function validateQuestions(lista = questions) {
  const ids = new Set();

  lista.forEach((q, index) => {
    const label = `Soal ${index + 1} (${q.id || 'tanpa ID'})`;

    if (!q.id) {
      console.warn(`${label}: ID kosong.`);
    } else if (ids.has(q.id)) {
      console.warn(`${label}: ID duplikat.`);
    }

    ids.add(q.id);

    // Soal baca kalimat tidak memakai kartu kata.
    if (q.tipe === 'baca') {
      if (!String(q.kalimat || '').trim()) {
        console.warn(`${label}: kalimat baca kosong.`);
      }

      return;
    }

    if (!Array.isArray(q.words) || !Array.isArray(q.correctOrder)) {
      console.warn(`${label}: words/correctOrder tidak valid.`);
      return;
    }

    if (q.correctOrder.length === 0) {
      console.warn(`${label}: correctOrder kosong.`);
      return;
    }

    const pool = [...q.words];

    q.correctOrder.forEach(word => {
      const pos = pool.indexOf(word);

      if (pos === -1) {
        console.warn(
          `${label}: kata "${word}" tidak ditemukan di words.`
        );
      } else {
        pool.splice(pos, 1);
      }
    });
  });
}

const SKOR_PER_PERCOBAAN = [100, 75, 50];
const SKOR_MINIMUM = 25;

function hitungSkorSoal(jumlahSalah) {
  return jumlahSalah < SKOR_PER_PERCOBAAN.length
    ? SKOR_PER_PERCOBAAN[jumlahSalah]
    : SKOR_MINIMUM;
}

function tentukanPredikat(nilai) {
  if (nilai >= 100) return 'Sempurna';
  if (nilai >= 90) return 'Sangat Baik';
  if (nilai >= 75) return 'Baik';
  if (nilai >= 60) return 'Cukup';
  return 'Perlu Pendampingan';
}

// Soal susun: 100/75/50/25 menurut jumlah percobaan salah.
// Soal baca : 100 dikurangi 5 poin tiap bacaan salah (pengenalan suara bisa keliru,
//             jadi pengurangannya lebih kecil).
function skorSoal(q, jumlahSalah) {
  return q.tipe === 'baca'
    ? BacaLogika.hitungPoin(jumlahSalah)
    : hitungSkorSoal(jumlahSalah);
}

function hitungHasilLevel() {
  const daftar = levelQuestions.map(q => {
    const state = getSoalState(q);

    return {
      soal_id: q.id,
      tipe: q.tipe || 'susun',
      jumlah_salah: state.jumlahSalah,
      skor: skorSoal(q, state.jumlahSalah),
      waktu_detik: state.waktuDetik
    };
  });

  const total = daftar.length;

  const nilai = total
    ? Math.round(
        daftar.reduce((totalSkor, item) => (
          totalSkor + item.skor
        ), 0) / total * 10
      ) / 10
    : 0;

  const langsungBenar = daftar.filter(
    item => item.jumlah_salah === 0
  ).length;

  return {
    daftar,
    total,
    nilai,
    langsungBenar,
    predikat: tentukanPredikat(nilai)
  };
}

async function kirimJson(url, body) {
  try {
    const response = await fetch(url, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify(body)
    });

    if (!response.ok) {
      console.warn(
        `Gagal mengirim laporan lokal ke ${url}: HTTP ${response.status}`
      );

      return false;
    }

    return true;
  } catch (error) {
    console.warn(`Server laporan lokal tidak tersedia: ${url}`, error);
    return false;
  }
}

function kirimHasilSoal(q, state) {

  });
}

const dashboardEventPromises = [];

// Kirim satu jawaban; diulang bila jaringan putus (event_id sama = aman diulang).
function kirimUlangEvent(event) {
  const kirimSekali = async () => {
    const response = await fetch(dashboardApiUrl, {
      method: 'POST',
      mode: 'cors',
      credentials: 'omit',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${ssoToken}`
      },
      body: JSON.stringify({
        ...event,
        ssoToken
      })
    });

    let body = {};

    try {
      body = await response.json();
    } catch (_) {
      body = {};
    }

    if (!response.ok || body.error) {
      const galat = new Error(
        body.error || `HTTP ${response.status} dari Dashboard`
      );

      // Penolakan server (4xx) tidak perlu diulang.
      galat.tetap = response.status >= 400 && response.status < 500;

      throw galat;
    }

    return {
      ok: true,
      data: body
    };
  };

  return (async () => {
    for (let percobaan = 1; ; percobaan++) {
      try {
        return await kirimSekali();
      } catch (error) {
        if (error.tetap || percobaan >= 3) throw error;

        await new Promise(r => setTimeout(r, 800 * percobaan));
      }
    }
  })().catch(error => {
    console.error('Jawaban gagal dikirim ke Dashboard:', error);

    return {
      ok: false,
      error: error.message || String(error)
    };
  });
}

const dashboardEventLog = [];

function kirimEventDashboard(event) {
  if (!hasDashboardSession || !dashboardApiUrl) {
    return Promise.resolve({
      ok: true,
      dilewati: true
    });
  }

  const promise = kirimUlangEvent(event);

  dashboardEventPromises.push(promise);
  dashboardEventLog.push({ event, promise });

  return promise;
}

async function selesaikanMisiDashboard(akhir = {}) {
  if (!hasDashboardSession || !dashboardApiUrl) {
    return {
      ok: true,
      dilewati: true
    };
  }

  // Pastikan pengiriman jawaban sebelumnya selesai; yang gagal dikirim ulang.
  for (const catatan of dashboardEventLog) {
    let hasilEvent = await catatan.promise;

    if (!hasilEvent.ok) {
      catatan.promise = kirimUlangEvent(catatan.event);
      hasilEvent = await catatan.promise;
    }

    if (!hasilEvent.ok) {
      return {
        ok: false,
        error: hasilEvent.error
      };
    }
  }

  try {
    const response = await fetch(dashboardApiUrl, {
      method: 'POST',
      mode: 'cors',
      credentials: 'omit',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${ssoToken}`
      },
      body: JSON.stringify({
        ssoToken,
        events: [],
        selesai: true,
        skor: akhir.skor,
        durasi_detik: akhir.durasi_detik
      })
    });

    let body = {};

    try {
      body = await response.json();
    } catch (_) {
      body = {};
    }

    if (
      !response.ok ||
      body.error ||
      body.selesai !== true
    ) {
      return {
        ok: false,
        error: body.error || (
          `Dashboard belum menyatakan misi selesai. HTTP ${response.status}`
        )
      };
    }

    console.info('Misi selesai tercatat di Dashboard.', body);

    return {
      ok: true,
      data: body
    };
  } catch (error) {
    console.error('Konfirmasi penyelesaian misi gagal:', error);

    return {
      ok: false,
      error: error.message || String(error)
    };
  }
}

function renderGambar(q) {
  const gambar = String(q.gambar || '').trim();

  soalGambarEl.innerHTML = '';
  disableImageZoom();

  if (!gambar) {
    soalGambarEl.innerText = '🖼️';
    return;
  }

  if (IMAGE_PATH_REGEX.test(gambar)) {
    const img = document.createElement('img');

    img.src = gambar;
    img.alt = String(q.instruksi || '').replace(/"/g, '');

      soalGambarEl.innerHTML = '';
      soalGambarEl.innerText = '🖼️';

      disableImageZoom();
    };

    soalGambarEl.appendChild(img);

    soalGambarEl.innerText = gambar;

  enableImageZoom(q, gambar);

// Hasil satu kali membaca (dinilai oleh baca-panel.js dan baca-logika.js).
async function tanganiHasilBaca(hasil) {
  const q = levelQuestions[currentQuestionIdx];

  if (levelTerkunci || !q || q.tipe !== 'baca') return;

  const state = getSoalState(q);
  const responseTimeMs = Math.max(0, Math.round(hasil.waktuMs));

  if (hasil.benar) {
    if (successDescEl) {
      successDescEl.innerText =
        `Bacaanmu tepat! Poin kalimat ini: ${BacaLogika.hitungPoin(state.jumlahSalah)}.`;
    }

    successModalEl.classList.remove('hidden');

    if (!state.selesai) {
      state.selesai = true;

      // Lampu level menyala untuk bacaan benar pertama pada soal ini.
      LenteraMisi.tandaiBenar(q.id);

      state.waktuDetik = Math.round((Date.now() - state.mulai) / 1000);

      kirimHasilSoal(q, state);
    }
  } else {
    // Hanya bacaan yang benar-benar terdengar dan belum tepat yang mengurangi poin.
    if (!state.selesai) {
      state.jumlahSalah++;
      BacaPanel.perbaruiPoin(state.jumlahSalah, true);
    }

    if (errorTitleEl) {
      errorTitleEl.innerText = `Belum tepat, poinmu berkurang ${BacaLogika.POIN_PER_SALAH}`;
    }

    modalErrorDescEl.innerText =
      `Yang terdengar: "${hasil.teks}". Baca lagi pelan-pelan, mulai dari kata yang bergaris merah. ` +
      `Poin kalimat ini sekarang ${BacaLogika.hitungPoin(state.jumlahSalah)}.`;

    errorModalEl.classList.remove('hidden');
  }

  await kirimEventDashboard({
    event_id: buatEventId(),
    item_id: q.id,
    question_id: q.id,
    kompetensi: KOMPETENSI_BACA,
    level: currentLevel,
    benar: hasil.benar,
    is_correct: hasil.benar,
    pola: hasil.pola || '',
    error_code: hasil.pola || '',
    error_pattern: hasil.pola || '',
    waktu_ms: responseTimeMs,
    response_time_ms: responseTimeMs,
    // Suara contoh atau bantuan guru dihitung sebagai petunjuk.
    petunjuk: Boolean(hasil.pakaiContoh || hasil.bantuanGuru),
    used_hint: Boolean(hasil.pakaiContoh || hasil.bantuanGuru),
    waktu: new Date().toISOString()
  });
}

BacaPanel.init({ onHasil: tanganiHasilBaca });

function renderWordBank() {
  wordBankEl.innerHTML = '';

  availableWords.forEach(word => {
    const btn = document.createElement('div');

    btn.className =
      'word-card interactive-word bg-white border-2 border-slate-200 ' +
      'hover:border-orange-400 font-semibold text-slate-700 px-4 py-2 ' +
      'rounded-xl shadow-sm text-xs cursor-pointer';

    btn.dataset.word = word;
    btn.innerText = word;

    btn.onclick = () => {
      if (grabbedWord) return;

      const index = availableWords.indexOf(word);
      if (index === -1) return;

      availableWords.splice(index, 1);
      selectedWords.push(word);

      renderWordBank();
      renderDropZone();
    };

    wordBankEl.appendChild(btn);
  });
}

function renderDropZone() {
  dropZoneEl.innerHTML = '';

  if (selectedWords.length === 0) {
    dropZoneEl.innerHTML =
      '<span class="text-xs text-slate-400 italic">' +
      'Cubit kata di bawah, seret, lalu lepaskan di sini...' +
      '</span>';

    return;
  }

  selectedWords.forEach((word, index) => {
    const chip = document.createElement('div');

    chip.className =
      'word-card interactive-cancel bg-orange-500 text-white ' +
      'font-semibold px-3 py-1.5 rounded-xl flex items-center gap-2 ' +
      'shadow-sm text-xs cursor-pointer';

    const wordSpan = document.createElement('span');
    wordSpan.textContent = word;

    const closeSpan = document.createElement('span');
    closeSpan.className = 'text-orange-200 font-bold text-xs';
    closeSpan.textContent = '✕';

    chip.appendChild(wordSpan);
    chip.appendChild(closeSpan);

    chip.onclick = () => {
      selectedWords.splice(index, 1);
      availableWords.push(word);

      renderWordBank();
      renderDropZone();
    };

    dropZoneEl.appendChild(chip);
  });
}

function countUpScore(target, duration = 1600) {
  const start = performance.now();

  function tick(now) {
    // Waktu frame bisa sedikit lebih awal dari `start`: jangan biarkan progres negatif (angka minus sesaat).
    const t = Math.max(0, Math.min((now - start) / duration, 1));
    const eased = 1 - Math.pow(1 - t, 3);

    resultScoreEl.innerText = Math.round(target * eased);

    if (t < 1) {
      requestAnimationFrame(tick);
    }
  }

  requestAnimationFrame(tick);
}

async function submitAnswer() {
  // Level terkunci (waktu habis) atau soal belum dimuat dari Dashboard.
  if (levelTerkunci || !levelQuestions[currentQuestionIdx]) {
    return;
  }

  // FIX: jangan proses jika salah satu popup jawaban masih terbuka.
  if (
    !errorModalEl.classList.contains('hidden') ||
    !successModalEl.classList.contains('hidden')
  ) {
    return;
  }

  const q = levelQuestions[currentQuestionIdx];
  const state = getSoalState(q);

  const responseTimeMs = Math.max(
    0,
    Date.now() - startTime
  );

  const userSentence = selectedWords.join(' ');

  const semuaJawabanBenar = [
    q.correctOrder,
    ...(Array.isArray(q.jawabanLain) ? q.jawabanLain : [])
  ];

  const kunci = arr => [...arr].sort().join('|');

  const isCorrect = q.urutanBebas
    ? semuaJawabanBenar.some(
        susunan => kunci(susunan) === kunci(selectedWords)
      )
    : semuaJawabanBenar.some(
        susunan => susunan.join(' ') === userSentence
      );

  let errorPattern = '';

  if (!isCorrect) {
    const userKey = selectedWords.join('_');

    if (q.urutanBebas) {
      const sisa = [...q.correctOrder];

      const semuaBenar = selectedWords.every(word => {
        const index = sisa.indexOf(word);

        if (index === -1) return false;

        sisa.splice(index, 1);
        return true;
      });

      errorPattern =
        (q.errorPatterns && q.errorPatterns[userKey]) ||
        (semuaBenar ? 'JAWABAN_KURANG' : 'PILIHAN_SALAH');

      modalErrorDescEl.innerText = semuaBenar
        ? `Jawabanmu: "${userSentence}". Pilihanmu sudah benar, tetapi masih ada yang kurang.`
        : `Jawabanmu: "${userSentence}". Ada pilihan yang tidak sesuai gambar. Coba lagi!`;

    } else {
      errorPattern =
        (q.errorPatterns && q.errorPatterns[userKey]) ||
        'URUTAN_TIDAK_TETAP';

      modalErrorDescEl.innerText =
        `Susunanmu: "${userSentence}" masih salah. Ayo coba lagi!`;
    }

    if (errorTitleEl) errorTitleEl.innerText = TEKS_SUSUN_SALAH;

    errorModalEl.classList.remove('hidden');

    if (!state.selesai && selectedWords.length > 0) {
      state.jumlahSalah++;
    }

  } else {
    if (successDescEl) successDescEl.innerText = TEKS_SUSUN_SUKSES;

    successModalEl.classList.remove('hidden');

    if (!state.selesai) {
      state.selesai = true;

      // Lampu level menyala untuk jawaban benar pertama pada soal ini.
      LenteraMisi.tandaiBenar(q.id);

      state.waktuDetik = Math.round(
        (Date.now() - state.mulai) / 1000
      );

      // Laporan lokal hanya dikirim ketika soal berhasil dijawab.
      kirimHasilSoal(q, state);
    }
  }

  // Kompetensi dari token diprioritaskan saat game dibuka
  // melalui Dashboard. Semua soal lama level itu digunakan.
  const kompetensiUntukDashboard = kompetensiEvent(q);

  await kirimEventDashboard({
    event_id: buatEventId(),
    item_id: q.id,
    question_id: q.id,
    kompetensi: kompetensiUntukDashboard,
    level: currentLevel,
    benar: isCorrect,
    is_correct: isCorrect,
    pola: errorPattern,
    error_code: errorPattern,
    error_pattern: errorPattern,
    waktu_ms: responseTimeMs,
    response_time_ms: responseTimeMs,
    petunjuk: false,
    used_hint: false,
    waktu: new Date().toISOString()
  });
}

const CFG = {
  detectConf: 0.6,
  trackConf: 0.5,
  filterMinCutoff: 0.8,
  filterBeta: 0.05,
  filterDCutoff: 1.0,
  smooth: 0.5,
  edgeMargin: 0.08,
  pinchOn: 0.50,
  pinchOff: 0.80,
  pinchOnFrames: 2,
  pinchOffFrames: 3,
  lostGraceFrames: 20,
  grabWindowMs: 700,
  grabPadding: 40,
  dropPadding: 40,
  hitRadius: 20,
  dwellMs: 1200,
  lockDistance: 60
};

let smoothY = null;

let pinchActive = false;
let pinchFrames = 0;
let openFrames = 0;
let lostFrames = 0;

let pinchStartAt = 0;

// ============================================================
// ONE EURO FILTER
// ============================================================

class OneEuro {
  constructor() {
    this.reset();
  }

  reset() {
    this.xPrev = null;
    this.dxPrev = 0;
    this.tPrev = 0;
  }

  static alpha(cutoff, dt) {
    const tau = 1 / (2 * Math.PI * cutoff);
    return 1 / (1 + tau / dt);
  }

  filter(x, t) {
    if (this.xPrev === null) {
      this.xPrev = x;
      this.dxPrev = 0;
      this.tPrev = t;

      return x;
    }

    const dt = Math.max(
      (t - this.tPrev) / 1000,
      0.001
    );

    this.tPrev = t;

    const dx = (x - this.xPrev) / dt;

    this.dxPrev +=
      OneEuro.alpha(CFG.filterDCutoff, dt) *
      (dx - this.dxPrev);

    const cutoff =
      CFG.filterMinCutoff +
      CFG.filterBeta * Math.abs(this.dxPrev);

    this.xPrev +=
      OneEuro.alpha(cutoff, dt) * (x - this.xPrev);

    return this.xPrev;
  }
}

const lmFilters = Array.from(
  { length: 21 },
  () => ({
    x: new OneEuro(),
    y: new OneEuro()
  })
);

function resetLandmarkFilters() {
  lmFilters.forEach(filter => {
    filter.x.reset();
    filter.y.reset();
  });
}

function smoothLandmarks(raw, vw, vh, t) {
  return raw.map((point, index) => ({
    x: lmFilters[index].x.filter(point.x * vw, t) / vw,
    y: lmFilters[index].y.filter(point.y * vh, t) / vh
  }));
}

const DOT_COLORS = {
  line: '#60a5fa',
  joint: '#ffffff',
  jointEdge: '#3b82f6',
  tipEdge: '#ffffff',
  open: '#f97316',
  near: '#fbbf24',
  pinch: '#10b981',
  shadow: 'rgba(15, 23, 42, 0.35)'
};

const HAND_CONNECTIONS = [
  [0, 1], [1, 2], [2, 3], [3, 4],
  [0, 5], [5, 6], [6, 7], [7, 8],
  [5, 9], [9, 10], [10, 11], [11, 12],
  [9, 13], [13, 14], [14, 15], [15, 16],
  [13, 17], [17, 18], [18, 19], [19, 20],
  [0, 17]
];

const FINGER_TIPS = [4, 8, 12, 16, 20];

  'z-index:2147483000;display:none;';

document.body.appendChild(handCanvas);

  'z-index:2147483001;display:none;';

function hideMarkers() {
  handCanvas.style.display = 'none';
  camCanvas.style.display = 'none';
}

function paintHand(ctx, pointAt, state, scale) {
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';
  ctx.shadowColor = DOT_COLORS.shadow;
  ctx.shadowBlur = 6 * scale;

  ctx.lineWidth = 4 * scale;
  ctx.strokeStyle = DOT_COLORS.line;
  ctx.beginPath();

  HAND_CONNECTIONS.forEach(([a, b]) => {
    const p = pointAt(a);
    const q = pointAt(b);

    ctx.moveTo(p.x, p.y);
    ctx.lineTo(q.x, q.y);
  });

  ctx.stroke();

  if (state !== 'open') {
    const thumb = pointAt(4);
    const index = pointAt(8);

    ctx.lineWidth = 5 * scale;
    ctx.strokeStyle =
      state === 'pinch'
        ? DOT_COLORS.pinch
        : DOT_COLORS.near;

    ctx.beginPath();
    ctx.moveTo(thumb.x, thumb.y);
    ctx.lineTo(index.x, index.y);
    ctx.stroke();
  }

  for (let id = 0; id < 21; id++) {
    const point = pointAt(id);
    const isTip = FINGER_TIPS.includes(id);

    let fill = DOT_COLORS.joint;

    if (id === 4 || id === 8) {
      if (state === 'pinch') {
        fill = DOT_COLORS.pinch;
      } else if (state === 'near') {
        fill = DOT_COLORS.near;
      } else {
        fill = DOT_COLORS.open;
      }
    }

    ctx.beginPath();

    ctx.arc(
      point.x,
      point.y,
      (isTip ? 13 : 7) * scale,
      0,
      Math.PI * 2
    );

    ctx.fillStyle = fill;
    ctx.fill();

    ctx.lineWidth = (isTip ? 3 : 2) * scale;

    ctx.strokeStyle =
      isTip
        ? DOT_COLORS.tipEdge
        : DOT_COLORS.jointEdge;

    ctx.stroke();
  }

  ctx.shadowBlur = 0;
}

function drawMarkers(landmarks, toScreen, state) {
  const dpr = window.devicePixelRatio || 1;
  const width = window.innerWidth;
  const height = window.innerHeight;

  if (
    handCanvas.width !== Math.round(width * dpr) ||
    handCanvas.height !== Math.round(height * dpr)
  ) {
    handCanvas.width = Math.round(width * dpr);
    handCanvas.height = Math.round(height * dpr);

    handCanvas.style.width = `${width}px`;
    handCanvas.style.height = `${height}px`;
  }

  handCanvas.style.display = 'block';

  handCtx.setTransform(dpr, 0, 0, dpr, 0, 0);
  handCtx.clearRect(0, 0, width, height);

  paintHand(
    handCtx,
    id => toScreen(landmarks[id]),
    state,
    1
  );

  const rect = videoElement.getBoundingClientRect();

  if (!rect.width || !rect.height) return;

  const w = Math.round(rect.width);
  const h = Math.round(rect.height);

  if (camCanvas.width !== w || camCanvas.height !== h) {
    camCanvas.width = w;
    camCanvas.height = h;
  }

  camCanvas.style.display = 'block';
  camCanvas.style.left = `${rect.left}px`;
  camCanvas.style.top = `${rect.top}px`;
  camCanvas.style.width = `${w}px`;
  camCanvas.style.height = `${h}px`;

  const vw = videoElement.videoWidth || 4;
  const vh = videoElement.videoHeight || 3;

  const scale = Math.max(w / vw, h / vh);

  const dw = vw * scale;
  const dh = vh * scale;

  const ox = (w - dw) / 2;
  const oy = (h - dh) / 2;

  camCtx.clearRect(0, 0, w, h);

  paintHand(
    camCtx,
    id => ({
      x: ox + (1 - landmarks[id].x) * dw,
      y: oy + landmarks[id].y * dh
    }),
    state,
    0.4
  );
}

function setCursorProgress(progress) {
  const px = Math.max(0, Math.min(1, progress)) * 10;

  virtualCursorEl.style.boxShadow =
    px > 0
      ? `0 0 0 ${px}px rgba(249,115,22,0.35)`
      : '';
}

function resetDwell() {
  if (hoverEl) {
    hoverEl.classList.remove('cursor-hover');
  }

  hoverEl = null;
  dwellStart = 0;

  setCursorProgress(0);
}

function findInteractiveNear(x, y) {
  const radius = CFG.hitRadius;

  const offsets = [
    [0, 0],
    [radius, 0],
    [-radius, 0],
    [0, radius],
    [0, -radius]
  ];

  for (const [dx, dy] of offsets) {
    const element = document.elementFromPoint(
      x + dx,
      y + dy
    );

    const hit = element
      ? element.closest('.interactive-elem, .interactive-cancel')
      : null;

    if (hit) return hit;
  }

  return null;
}

function findWordNear(x, y) {
  const padding = CFG.grabPadding;

  let best = null;
  let bestDistance = Infinity;

  document.querySelectorAll('.interactive-word').forEach(element => {
    const rect = element.getBoundingClientRect();

    if (
      x >= rect.left - padding &&
      x <= rect.right + padding &&
      y >= rect.top - padding &&
      y <= rect.bottom + padding
    ) {
      const distance = Math.hypot(
        x - (rect.left + rect.width / 2),
        y - (rect.top + rect.height / 2)
      );

      if (distance < bestDistance) {
        bestDistance = distance;
        best = element;
      }
    }
  });

  return best;
}

function isInsideDropZone(x, y) {
  const rect = dropZoneEl.getBoundingClientRect();
  const padding = CFG.dropPadding;

  return (
    x >= rect.left - padding &&
    x <= rect.right + padding &&
    y >= rect.top - padding &&
    y <= rect.bottom + padding
  );
}

function updateDwell(x, y, now) {
  if (dwellLock) {
    const distance = Math.hypot(
      x - dwellLock.x,
      y - dwellLock.y
    );

    if (distance < CFG.lockDistance) {
      resetDwell();
      return;
    }

    dwellLock = null;
  }

  const target = findInteractiveNear(x, y);

  if (!target) {
    resetDwell();
    return;
  }

  if (target !== hoverEl) {
    resetDwell();

    hoverEl = target;
    hoverEl.classList.add('cursor-hover');

    dwellStart = now;
    return;
  }

  const progress = (now - dwellStart) / CFG.dwellMs;

  setCursorProgress(progress);

  if (progress >= 1) {
    const targetClick = hoverEl;

    dwellLock = { x, y };

    resetDwell();

    targetClick.click();
  }
}

function dropGrabbedWord(x, y) {
  if (!grabbedWord) return;

  if (isInsideDropZone(x, y)) {
    selectedWords.push(grabbedWord);
  } else {
    availableWords.push(grabbedWord);
  }

  grabbedWord = null;
  floatingWordEl.style.display = 'none';

  renderWordBank();
  renderDropZone();
}

function onResults(results) {
  const now = performance.now();

  if (
    !results.multiHandLandmarks ||
    results.multiHandLandmarks.length === 0
  ) {
    lostFrames++;

    if (lostFrames > CFG.lostGraceFrames) {
      gestureStatusEl.innerText = 'Mencari Tangan...';

      cameraDotEl.className =
        'w-1.5 h-1.5 rounded-full bg-amber-400 animate-ping';

      virtualCursorEl.style.display = 'none';
      floatingWordEl.style.display = 'none';

      hideMarkers();
      resetDwell();

      pinchActive = false;
      pinchFrames = 0;
      openFrames = 0;

      smoothX = null;
      smoothY = null;

      resetLandmarkFilters();

      if (grabbedWord) {
        availableWords.push(grabbedWord);
        grabbedWord = null;

        renderWordBank();
        renderDropZone();
      }
    }

    return;
  }

  lostFrames = 0;

  const vw = videoElement.videoWidth || 4;
  const vh = videoElement.videoHeight || 3;

  const landmarks = smoothLandmarks(
    results.multiHandLandmarks[0],
    vw,
    vh,
    now
  );

  const thumbTip = landmarks[4];
  const indexTip = landmarks[8];
  const wrist = landmarks[0];
  const middleMcp = landmarks[9];

  const distance = (a, b) => Math.hypot(
    (a.x - b.x) * vw,
    (a.y - b.y) * vh
  );

  const handSize = Math.max(
    distance(wrist, middleMcp),
    0.0001
  );

  const ratio = distance(thumbTip, indexTip) / handSize;

  if (ratio < CFG.pinchOn) {
    pinchFrames++;
    openFrames = 0;
  } else if (ratio > CFG.pinchOff) {
    openFrames++;
    pinchFrames = 0;
  } else {
    pinchFrames = 0;
    openFrames = 0;
  }

  const wasPinching = pinchActive;

  if (
    !pinchActive &&
    pinchFrames >= CFG.pinchOnFrames
  ) {
    pinchActive = true;
    pinchStartAt = now;
  }

  if (
    pinchActive &&
    openFrames >= CFG.pinchOffFrames
  ) {
    pinchActive = false;
  }

  const margin = CFG.edgeMargin;

  const toScreen = point => ({
    x: Math.min(
      1,
      Math.max(
        0,
        ((1 - point.x) - margin) / (1 - 2 * margin)
      )
    ) * window.innerWidth,

    y: Math.min(
      1,
      Math.max(
        0,
        (point.y - margin) / (1 - 2 * margin)
      )
    ) * window.innerHeight
  });

  const thumbScreen = toScreen(thumbTip);
  const indexScreen = toScreen(indexTip);

  const usePinchPoint =
    pinchActive || ratio < CFG.pinchOn;

  const targetX = usePinchPoint
    ? (thumbScreen.x + indexScreen.x) / 2
    : indexScreen.x;

  const targetY = usePinchPoint
    ? (thumbScreen.y + indexScreen.y) / 2
    : indexScreen.y;

  if (smoothX === null) {
    smoothX = targetX;
    smoothY = targetY;
  } else {
    smoothX += (targetX - smoothX) * CFG.smooth;
    smoothY += (targetY - smoothY) * CFG.smooth;
  }

  const screenX = smoothX;
  const screenY = smoothY;

  if (pinchActive && openFrames === 1) {
    releasePos = { x: screenX, y: screenY };
  }

  if (pinchActive && openFrames === 0) {
    releasePos = null;
  }

  const state = pinchActive
    ? 'pinch'
    : (ratio < CFG.pinchOff ? 'near' : 'open');

  cameraDotEl.className =
    'w-1.5 h-1.5 rounded-full bg-emerald-500 animate-pulse';

  gestureStatusEl.innerText = pinchActive
    ? 'Mencubit 🤏'
    : (
        state === 'near'
          ? 'Hampir cubit...'
          : 'Telunjuk Aktif 👆'
      );

  virtualCursorEl.style.display = 'flex';
  virtualCursorEl.style.left = `${screenX}px`;
  virtualCursorEl.style.top = `${screenY}px`;
  virtualCursorEl.innerText = pinchActive ? '🤏' : '👆';

  drawMarkers(landmarks, toScreen, state);

  if (grabbedWord) {
    floatingWordEl.style.display = 'block';
    floatingWordEl.style.left = `${screenX}px`;
    floatingWordEl.style.top = `${screenY - 25}px`;
    floatingWordEl.innerText = `🤏 ${grabbedWord}`;
  } else {
    floatingWordEl.style.display = 'none';
  }

  // Ambil kata ketika mencubit.
  if (
    pinchActive &&
    !grabbedWord &&
    !isImageZoomOpen() &&
    now - pinchStartAt <= CFG.grabWindowMs
  ) {
    const target = findWordNear(screenX, screenY);

    if (target) {
      const word = target.dataset.word;
      const index = availableWords.indexOf(word);

      if (index !== -1) {
        availableWords.splice(index, 1);
        grabbedWord = word;

        renderWordBank();
        renderDropZone();
      }
    }
  }

  // Lepas kata.
  if (!pinchActive && wasPinching && grabbedWord) {
    const position = releasePos || {
      x: screenX,
      y: screenY
    };

    dropGrabbedWord(position.x, position.y);
    releasePos = null;
  }

  // Klik tombol dengan telunjuk saat tidak membawa kata.
  if (
    pinchActive ||
    grabbedWord ||
    ratio < CFG.pinchOn
  ) {
    resetDwell();
  } else {
    updateDwell(screenX, screenY, now);
  }
}

try {
  if (
    typeof Hands === 'undefined' ||
    typeof Camera === 'undefined' ||
    !videoElement
  ) {
    throw new Error('MediaPipe atau elemen kamera tidak tersedia.');
  }

  const hands = new Hands({
    locateFile: file =>
      `https://cdn.jsdelivr.net/npm/@mediapipe/hands/${file}`
  });

  hands.setOptions({
    maxNumHands: 1,
    modelComplexity: 1,
    minDetectionConfidence: CFG.detectConf,
    minTrackingConfidence: CFG.trackConf
  });

  hands.onResults(onResults);

  const camera = new Camera(videoElement, {
    onFrame: async () => {
      await hands.send({ image: videoElement });
    },
    width: 640,
    height: 480
  });

  camera.start().catch(error => {
    console.warn('Kamera tidak tersedia:', error);

    gestureStatusEl.innerText = 'Mode Sentuh Saja';

    cameraDotEl.className =
      'w-1.5 h-1.5 rounded-full bg-rose-500';
  });

} catch (error) {
  console.warn('Sensor gestur tidak tersedia:', error);

  if (gestureStatusEl) {
    gestureStatusEl.innerText = 'Mode Sentuh Aktif';
  }
}

async function mulaiPermainan() {
  if (!hasDashboardSession) {
    renderQuestion();
    return;
  }

  // Mode Dashboard: soal level ini berasal dari bank guru (API Dashboard).
  const data = await LenteraMisi.muat();

  if (!data) return;

  levelQuestions = data.soal;
  validateQuestions(levelQuestions);

  // Lanjutkan dari soal pertama yang belum dijawab benar.
  currentQuestionIdx = Math.max(
    0,
    levelQuestions.findIndex(q => !LenteraMisi.benar.has(q.id))
  );

  const boleh = await LenteraMisi.mulai({
    // Waktu habis: denda dikurangkan ke nilai akhir (lihat showResult).
    onGagal: () => {
      levelTerkunci = true;
      BacaPanel.batalkan();
      successModalEl.classList.add('hidden');
      errorModalEl.classList.add('hidden');
      resultModalEl.classList.add('hidden');
    },
    skor: () => hitungHasilLevel().nilai
  });

  if (!boleh) return;

  renderQuestion();
}
